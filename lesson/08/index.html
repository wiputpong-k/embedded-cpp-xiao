State machine

25 นาที · ฮาร์ดแวร์: ปุ่มกดที่ D1

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย ใช้ปุ่มที่ debounce แล้วสลับโหมดการกระพริบของ LED ด้วย state machine แบบ non-blocking

state machine คือการออกแบบโปรแกรมให้อยู่ใน สถานะ (state) ใดสถานะหนึ่งเสมอ และเปลี่ยนสถานะเมื่อเกิด เหตุการณ์ (event) เท่านั้น บทนี้มี 3 สถานะ คือ Off, Slow (กระพริบทุก 500 ms) และ Fast (ทุก 100 ms) เหตุการณ์เดียวคือ Event::Pressed จาก Debouncer ในบทที่แล้ว

ใช้ enum class Mode (บท 03) เก็บสถานะ และแยกกฎการเปลี่ยนสถานะไว้ในฟังก์ชัน next() ที่เดียว เวลาจะเพิ่มโหมดใหม่จึงรู้ทันทีว่าต้องแก้ตรงไหน

Hands-on

  1. สร้าง sketch ใหม่ชื่อ button_modes
  2. คัดลอกโฟลเดอร์ src จาก sketch button_count มาไว้ในโฟลเดอร์ของ sketch ใหม่ (ใช้ Sketch → Show Sketch Folder เปิดทั้งสองโฟลเดอร์)
  3. วางโค้ดข้างล่างใน button_modes.ino แล้วกด Upload
  4. กดปุ่ม: LED เขียวต้องเปลี่ยนจากดับ → กระพริบช้า → กระพริบเร็ว → ดับ วนไปเรื่อย ๆ
button_modes.ino
#include <Adafruit_TinyUSB.h>   // เปิด USB และ Serial (core นี้ต้องมี)
#include "src/Debouncer.h"

constexpr uint8_t kButtonPin = D1;
enum class Mode : uint8_t { Off, Slow, Fast };

Debouncer button;
Mode mode = Mode::Off;
uint32_t lastToggle = 0;
bool ledLit = false;

Mode next(Mode m) {
  switch (m) {
    case Mode::Off:  return Mode::Slow;
    case Mode::Slow: return Mode::Fast;
    case Mode::Fast: return Mode::Off;
  }
  return Mode::Off;
}

uint32_t periodMs(Mode m) { return m == Mode::Slow ? 500 : 100; }

void setup() {
  pinMode(kButtonPin, INPUT_PULLUP);
  pinMode(LED_GREEN, OUTPUT);
  digitalWrite(LED_GREEN, HIGH);
}

void loop() {
  uint32_t now = millis();
  bool pressed = (digitalRead(kButtonPin) == LOW);
  if (button.update(pressed, now) == Debouncer::Event::Pressed) {
    mode = next(mode);
  }
  if (mode == Mode::Off) {
    digitalWrite(LED_GREEN, HIGH);
  } else if (now - lastToggle >= periodMs(mode)) {
    lastToggle = now;
    ledLit = !ledLit;
    digitalWrite(LED_GREEN, ledLit ? LOW : HIGH);
  }
}

สังเกตว่าการกระพริบใช้ now - lastToggle >= periodMs(mode) แทน delay() loop จึงวนเร็วตลอดเวลา และตอบสนองต่อปุ่มได้ทันทีแม้อยู่ในโหมด Slow

เช็กความเข้าใจ

ทำไม switch มี return Mode::Off ท้ายฟังก์ชันอีก?

บาง compiler เตือนว่า control reaches end of non-void function แม้จะครอบทุก case แล้ว ใส่ไว้เพื่อความปลอดภัยและ warning-free

ถ้าจะเพิ่มโหมด "ติดค้าง" ต้องแก้ตรงไหนบ้าง?

เพิ่มชื่อใน enum class Mode, เพิ่ม case ใน next() ให้เข้าและออกจากโหมดใหม่ และเพิ่มเงื่อนไขใน loop ให้เขียน LOW ค้างไว้ ส่วนอื่นไม่ต้องแก้
