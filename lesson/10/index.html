src/RingBuffer.h
#pragma once
#include <stddef.h>

// core ของ Seeed nRF52 มี class RingBuffer อยู่แล้ว จึงต้องอยู่ใน namespace ไม่ให้ชื่อชนกัน
namespace xiao {

// FIFO ขนาดคงที่ ไม่ใช้ heap — ขนาด N ถูกกำหนดตอน compile
template <typename T, size_t N>
class RingBuffer {
  static_assert(N > 0, "N must be > 0");

public:
  bool push(const T& v) {
    if (count_ == N) return false;  // เต็ม
    buf_[head_] = v;
    head_ = (head_ + 1) % N;
    ++count_;
    return true;
  }

  bool pop(T& out) {
    if (count_ == 0) return false;  // ว่าง
    out = buf_[tail_];
    tail_ = (tail_ + 1) % N;
    --count_;
    return true;
  }

  // เต็มแล้วให้ทิ้งตัวเก่าสุด (เหมาะกับ moving average)
  void pushOverwrite(const T& v) {
    if (count_ == N) {
      T dropped;
      pop(dropped);
    }
    push(v);
  }

  size_t size() const { return count_; }
  bool full() const { return count_ == N; }
  const T& at(size_t i) const { return buf_[(tail_ + i) % N]; }  // 0 = เก่าสุด

private:
  T buf_[N]{};
  size_t head_ = 0, tail_ = 0, count_ = 0;
};

template <typename T, size_t N>
float average(const RingBuffer<T, N>& rb) {
  if (rb.size() == 0) return 0.0f;
  float sum = 0;
  for (size_t i = 0; i < rb.size(); ++i) sum += rb.at(i);
  return sum / rb.size();
}

}  // namespace xiao
moving_average.ino
#include <Adafruit_TinyUSB.h>   // เปิด USB และ Serial (core นี้ต้องมี)
#include "src/RingBuffer.h"

xiao::RingBuffer<uint16_t, 16> samples;   // 16 ค่าล่าสุด

void setup() { Serial.begin(115200); }

void loop() {
  uint16_t raw = analogRead(A0);
  samples.pushOverwrite(raw);
  Serial.print("raw:"); Serial.print(raw);
  Serial.print(" avg:"); Serial.println(average(samples));
  delay(20);
}

ทำไมต้องเขียน xiao::RingBuffer? core ของ Seeed nRF52 มี class ชื่อ RingBuffer อยู่แล้ว (ใช้ภายในตัวจัดการ UART) ถ้าประกาศชื่อเดียวกันไว้นอก namespace จะ compile ไม่ผ่านเพราะชื่อชนกัน ไฟล์ RingBuffer.h ของคอร์สจึงอยู่ใน namespace xiao นี่คือเหตุผลจริงที่ต้องมี namespace ตามบท 03 ส่วน average(samples) เรียกได้โดยไม่ต้องเติม xiao:: เพราะ compiler หาฟังก์ชันใน namespace ของชนิดพารามิเตอร์ให้เอง (argument-dependent lookup)

เปิด Serial Plotter เทียบเส้น raw กับ avg ลองเปลี่ยน N เป็น 4 และ 64 สังเกตความหน่วงและความเรียบ (moving average N จุด = delay ราว N/2 sample)

เช็กความเข้าใจ

ถ้าใช้ RingBuffer<uint16_t,16> และ RingBuffer<float,16> flash จะเป็นอย่างไร?

compiler สร้างโค้ด 2 ชุด (template instantiation) flash เพิ่มขึ้น ถ้ามีหลายชนิดเกินไปเรียกว่า code bloat

ถ้า ISR เป็นคน push และ loop เป็นคน pop ปลอดภัยไหม?

ยังไม่ปลอดภัย เพราะ count_ ถูกแก้จากทั้งสองฝั่ง ต้องครอบด้วย CriticalSection หรือออกแบบ SPSC ที่ฝั่ง producer แก้แค่ head ฝั่ง consumer แก้แค่ tail
