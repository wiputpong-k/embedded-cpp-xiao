Template: RingBuffer<T, N>

30 นาที · ฮาร์ดแวร์: Pot ที่ A0

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย เขียน container ขนาดคงที่ใช้ซ้ำได้ทุกชนิด โดยไม่ใช้ heap

ค่าที่อ่านจาก pot จะแกว่งเล็กน้อยทุกครั้ง วิธีทำให้เรียบคือเก็บค่าล่าสุด N ค่าไว้แล้วเฉลี่ย (moving average) ต้องมีที่เก็บที่ใส่ค่าใหม่และทิ้งค่าเก่าได้เรื่อย ๆ และอยากใช้ซ้ำได้ทั้งกับ uint16_t, float หรือ struct คำศัพท์ในบทนี้:

  • template แม่แบบของ class หรือฟังก์ชันที่เว้นชนิด (T) หรือค่าคงที่ (N) ไว้ เมื่อเขียน RingBuffer<uint16_t, 16> compiler จะสร้างโค้ดฉบับสำหรับ uint16_t 16 ช่องให้ เรียกว่า template instantiation
  • ring buffer / FIFO array ขนาดคงที่ที่ใช้วนรอบ: head คือช่องที่จะเขียนถัดไป tail คือช่องที่จะอ่านถัดไป ถึงท้าย array ก็วนกลับไปช่อง 0 ด้วย % N ค่าที่เข้าก่อนออกก่อน (First In, First Out)
  • heap หน่วยความจำที่จองตอนโปรแกรมทำงานด้วย new หรือ malloc buffer ในบทนี้ไม่ใช้ heap ขนาดรู้ตั้งแต่ตอน build (บท 11 อธิบายว่าทำไมสำคัญ)
  • static_assert เงื่อนไขที่ compiler ตรวจตอน compile ถ้าไม่จริงจะ build ไม่ผ่าน
C

ทำ buffer สำหรับ int, float แยกกัน หรือใช้ void* + macro ซึ่งไม่มี type safety

C++

template ให้ compiler สร้างโค้ดเฉพาะชนิดให้ ขนาด N เป็นค่าตอน compile ตรวจได้ด้วย static_assert

Hands-on

sketch นี้พิสูจน์ว่า template ตัวเดียวสร้าง buffer ตามชนิดและขนาดที่สั่งได้ ขนาดถูกล็อกตั้งแต่ตอน build และไม่ใช้ heap ใช้วงจร pot จากบท 06

  1. สร้าง sketch ใหม่ บันทึกชื่อ moving_average
  2. สร้างไฟล์ src/RingBuffer.h แบบเดียวกับบท 05 (Sketch → Show Sketch Folder → สร้างโฟลเดอร์ src → สร้างไฟล์ RingBuffer.h) แล้ววางโค้ดชุดแรกข้างล่าง
  3. วางโค้ดชุดที่สองลงใน moving_average.ino กด Upload แล้วจดตัวเลข Global variables use ในแถบ Output ไว้
  4. เปิด Tools → Serial Plotter ที่ 115200 baud แล้วหมุน pot
src/RingBuffer.h
#pragma once
#include <stddef.h>

// core ของ Seeed nRF52 มี class RingBuffer อยู่แล้ว จึงต้องอยู่ใน namespace ไม่ให้ชื่อชนกัน
namespace xiao {

// FIFO ขนาดคงที่ ไม่ใช้ heap — ขนาด N ถูกกำหนดตอน compile
template <typename T, size_t N>
class RingBuffer {
  static_assert(N > 0, "N must be > 0");

public:
  bool push(const T& v) {
    if (count_ == N) return false;  // เต็ม
    buf_[head_] = v;
    head_ = (head_ + 1) % N;
    ++count_;
    return true;
  }

  bool pop(T& out) {
    if (count_ == 0) return false;  // ว่าง
    out = buf_[tail_];
    tail_ = (tail_ + 1) % N;
    --count_;
    return true;
  }

  // เต็มแล้วให้ทิ้งตัวเก่าสุด (เหมาะกับ moving average)
  void pushOverwrite(const T& v) {
    if (count_ == N) {
      T dropped;
      pop(dropped);
    }
    push(v);
  }

  size_t size() const { return count_; }
  bool full() const { return count_ == N; }
  const T& at(size_t i) const { return buf_[(tail_ + i) % N]; }  // 0 = เก่าสุด

private:
  T buf_[N]{};
  size_t head_ = 0, tail_ = 0, count_ = 0;
};

template <typename T, size_t N>
float average(const RingBuffer<T, N>& rb) {
  if (rb.size() == 0) return 0.0f;
  float sum = 0;
  for (size_t i = 0; i < rb.size(); ++i) sum += rb.at(i);
  return sum / rb.size();
}

}  // namespace xiao
moving_average.ino
#include <Adafruit_TinyUSB.h>   // เปิด USB และ Serial (core นี้ต้องมี)
#include "src/RingBuffer.h"

xiao::RingBuffer<uint16_t, 16> samples;   // 16 ค่าล่าสุด

void setup() { Serial.begin(115200); }

void loop() {
  uint16_t raw = analogRead(A0);
  samples.pushOverwrite(raw);
  Serial.print("raw:"); Serial.print(raw);
  Serial.print(" avg:"); Serial.println(average(samples));
  delay(20);
}

ผลที่ควรเห็น: เส้น raw ขยับตาม pot และมีหยักเล็ก ๆ ส่วนเส้น avg เรียบกว่าและตามมาช้ากว่าเล็กน้อย

ผลนี้บอกอะไร ลองทีละข้อ:

  • ขนาดล็อกตั้งแต่ build เปลี่ยน 16 เป็น 64 แล้วกด Verify ตัวเลข Global variables use จะเพิ่ม 96 byte พอดี (เพิ่ม 48 ช่อง × 2 byte ของ uint16_t) เพราะ buffer เป็นตัวแปร global ที่ linker จองที่ไว้ตายตัว (บท 01) ไม่ได้จองจาก heap ตอนรัน
  • N มากขึ้น = เรียบขึ้นแต่ช้าลง Upload ค่า 64 แล้วดู Plotter เส้น avg เรียบขึ้นแต่ตามช้าลง (moving average N จุด หน่วงสัญญาณราว N/2 ค่า)
  • static_assert เปลี่ยนเป็น 0 แล้วกด Verify จะได้ error: static assertion failed: N must be > 0 คือบั๊กถูกจับตั้งแต่ตอน compile จากนั้นเปลี่ยนกลับเป็น 16

ทำไมต้องเขียน xiao::RingBuffer? core ของ Seeed nRF52 มี class ชื่อ RingBuffer อยู่แล้ว ถ้าประกาศชื่อเดียวกันไว้นอก namespace จะ compile ไม่ผ่านเพราะชื่อชนกัน นี่คือเหตุผลจริงที่ต้องมี namespace ตามบท 03 ส่วน average(samples) ไม่ต้องเติม xiao:: เพราะ compiler หาฟังก์ชันใน namespace เดียวกับชนิดของพารามิเตอร์ให้เอง

เช็กความเข้าใจ

ถ้าใช้ RingBuffer<uint16_t,16> และ RingBuffer<float,16> flash จะเป็นอย่างไร?

compiler สร้างโค้ด 2 ชุด ชุดละชนิด flash จึงเพิ่มขึ้น ถ้าใช้หลายชนิดมากเกินไป flash จะบวมโดยไม่รู้ตัว เรียกว่า code bloat

ถ้า ISR (บท 12) เป็นคน push และ loop เป็นคน pop ปลอดภัยไหม?

ยังไม่ปลอดภัย เพราะ count_ ถูกแก้จากทั้งสองฝั่ง ISR อาจแทรกเข้ามากลางการแก้ค่า ต้องครอบด้วย CriticalSection (บท 06) หรือออกแบบใหม่ให้ฝั่ง ISR แก้แค่ head และฝั่ง loop แก้แค่ tail
