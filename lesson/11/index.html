Hands-on

memmap.ino
uint8_t bigStatic[8192];        // .bss  → นับใน "Global variables use"
const char kMsg[] = "hello";    // .rodata → อยู่ใน flash

void setup() {
  Serial.begin(115200);
  while (!Serial) {}
  uint8_t onStack[256];         // stack → ไม่อยู่ในตัวเลขตอน compile
  onStack[0] = 1;
  bigStatic[0] = onStack[0];
  Serial.print("flash  : 0x"); Serial.println((uintptr_t)kMsg, HEX);
  Serial.print("static : 0x"); Serial.println((uintptr_t)bigStatic, HEX);
  Serial.print("stack  : 0x"); Serial.println((uintptr_t)onStack, HEX);
}

void loop() {}

ดูว่า address ขึ้นต้น 0x000… (flash) หรือ 0x2000… (RAM) ตามรูปไหม จากนั้นเปลี่ยน bigStatic เป็น 32768 แล้วดูตัวเลข RAM ตอน compile เปลี่ยน

กฎที่ใช้ในทีมจริง

  • จอง memory ตอน init เท่านั้น หลังจากนั้นห้าม new/malloc ใน loop เพราะ heap fragment แล้วพังหลังรันหลายวัน
  • เลี่ยง String ของ Arduino ใช้ char buf[32] + snprintf แทน
  • ใช้ std::array / buffer ขนาดคงที่ หลีกเลี่ยง std::vector, std::function ที่อาจจอง heap เอง
  • embedded มักปิด exceptions และ RTTI (-fno-exceptions -fno-rtti) จึงจัดการ error ด้วยค่าที่ return

เช็กความเข้าใจ

ตัวแปร static ภายในฟังก์ชันอยู่ส่วนไหน?

.data หรือ .bss เหมือน global แต่ scope จำกัดในฟังก์ชัน จึงคงค่าระหว่างการเรียก

stack overflow บน MCU ไม่มี OS แสดงอาการอย่างไร?

stack โตลงไปทับ heap หรือ .bss ทำให้ค่าตัวแปรเพี้ยนแบบสุ่มหรือ HardFault วิธีตรวจคือ stack painting (เติม pattern แล้วดูว่าถูกเขียนลึกแค่ไหน) หรือเปิด MPU guard

มุมงานจริง

ใช้ memory map นี้ทุกครั้งที่เจอ HardFault หรือ RAM ไม่พอ: ดูว่า address ที่พังอยู่ใน .text / .data / .bss / heap / stack ส่วนไหน และเปิดไฟล์ .map จาก linker ดูว่าอะไรกินที่มากที่สุด
