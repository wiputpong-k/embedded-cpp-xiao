Inheritance, virtual และ interface

25 นาที · ฮาร์ดแวร์: Pot ที่ A0

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย ออกแบบ interface ISensor ให้สลับเซนเซอร์จริงกับเซนเซอร์ปลอมได้ และรู้ต้นทุนของ virtual

ใน C ทำ polymorphism ด้วย struct ของ function pointer เอง ใน C++ ใช้ virtual แล้ว compiler สร้างตาราง (vtable) ให้ ประโยชน์หลักใน embedded คือสลับ implementation ได้ เช่น ใช้ FakeSensor ตอน test โดยไม่ต้องแก้โค้ดที่เรียกใช้

Hands-on

sensors.ino
class ISensor {
public:
  virtual ~ISensor() = default;
  virtual const char* name() const = 0;   // pure virtual = ต้อง override
  virtual float read() = 0;
};

class PotSensor : public ISensor {
public:
  explicit PotSensor(uint8_t pin) : pin_(pin) {}
  const char* name() const override { return "pot"; }
  float read() override { return analogRead(pin_) * (100.0f / 1023.0f); }  // %
private:
  uint8_t pin_;
};

class FakeSensor : public ISensor {
public:
  const char* name() const override { return "fake"; }
  float read() override { t_ += 0.1f; return 50.0f + 40.0f * sinf(t_); }
private:
  float t_ = 0;
};

PotSensor pot(A0);
FakeSensor fake;
ISensor* sensors[] = {&pot, &fake};

void setup() { Serial.begin(115200); }

void loop() {
  for (ISensor* s : sensors) {
    Serial.print(s->name()); Serial.print(':');
    Serial.print(s->read()); Serial.print(' ');
  }
  Serial.println();
  delay(50);
}

เปิด Tools → Serial Plotter จะเห็นกราฟ 2 เส้น หมุน pot เทียบกับคลื่น sine

เช็กความเข้าใจ

ทำไม base class ต้องมี virtual destructor?

ถ้า delete object ผ่าน pointer ของ base แล้ว destructor ไม่ virtual destructor ของ derived จะไม่ถูกเรียก resource รั่ว

ถ้าไม่อยากจ่ายต้นทุน virtual มีทางเลือกอะไร?

ใช้ template (static polymorphism / CRTP) ให้ compiler รู้ชนิดตอน compile แล้ว inline ได้ แลกกับ code size ที่อาจเพิ่มขึ้น

มุมงานจริง

เวลาทีมถกกันว่า "virtual ช้าไหม" ให้คุยด้วยตัวเลข: เพิ่ม load 2 ครั้ง + indirect branch ต่อ call บน Cortex-M4 ราวไม่กี่ cycle ต้นทุนจริงคือ inline ไม่ได้ จึงควรเลี่ยงเฉพาะใน hot loop หรือ ISR
