Mini project 2: รับคำสั่งทาง Serial

25 นาที · ฮาร์ดแวร์: LED + ปุ่ม + Pot

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย เพิ่มการรับคำสั่งจาก Serial Monitor แบบไม่ block ให้ firmware จากบทที่แล้ว

firmware จริงมักต้องรับคำสั่งจากภายนอก เช่น ตั้งค่า หรือสั่งให้ทดสอบ ปัญหาคือข้อความมาถึงทีละตัวอักษร ถ้ารอให้ครบทั้งบรรทัดด้วยฟังก์ชันอย่าง Serial.readStringUntil() loop จะค้างระหว่างรอ task อื่นจะหยุดทำงาน

handleSerial() แก้ปัญหานี้โดยอ่านเฉพาะตัวอักษรที่มาถึงแล้ว เก็บสะสมใน buffer แล้วออกจากฟังก์ชันทันที เมื่อเจอตัวขึ้นบรรทัดใหม่ (\n หรือ \r) จึงแปลงข้อความเป็นตัวเลขแล้วใช้งาน

  • static ทำให้ buf และ len จำค่าข้ามการเรียกแต่ละครั้ง (บท 11) ข้อความครึ่งบรรทัดจึงไม่หาย
  • len < sizeof(buf) - 1 กันไม่ให้เขียนเกินขนาด buffer และเว้นที่ไว้ 1 ช่องสำหรับ \0
  • ตรวจช่วงค่า 50–5000 ก่อนใช้เสมอ ข้อมูลจากภายนอกต้องไม่ถูกเชื่อโดยไม่ตรวจ

Hands-on

  1. เปิด sketch firmware จากบทที่แล้ว แล้วแทนที่โค้ดทั้งหมดด้วยโค้ดข้างล่าง (มีฟังก์ชัน handleSerial() เพิ่ม และเรียกใช้ในบรรทัดสุดท้ายของ loop())
  2. กด Upload แล้วเปิด Tools → Serial Monitor ตั้งความเร็ว 115200 baud
  3. ที่มุมขวาของ Serial Monitor มีตัวเลือกการขึ้นบรรทัด ให้เลือก New Line ไม่อย่างนั้นบอร์ดจะไม่รู้ว่าพิมพ์จบบรรทัดแล้ว
  4. พิมพ์ 100 ในช่องข้อความด้านบนของ Serial Monitor แล้วกด Enter LED เขียวต้องกระพริบเร็วขึ้น ลองพิมพ์ 2000 แล้วลอง 10 (นอกช่วง ต้องไม่มีอะไรเปลี่ยน)
firmware/firmware.ino
#include "src/Led.h"
#include "src/Debouncer.h"
#include "src/RingBuffer.h"

#if __has_include("src/version.h")
#include "src/version.h"          // CI สร้างไฟล์นี้ (บท 23)
#else
#define FW_VERSION "dev"
#endif

constexpr uint8_t kButtonPin = D1;
constexpr uint8_t kPotPin    = A0;

Led status(LED_GREEN, true);
Led ext(D2, false);
Debouncer button;
RingBuffer<uint16_t, 16> samples;
bool paused = false;

struct Task {
  uint32_t periodMs;
  uint32_t lastRunMs;
  void (*run)();
};

void blinkTask() { if (!paused) status.toggle(); }

void buttonTask() {
  bool pressed = digitalRead(kButtonPin) == LOW;
  if (button.update(pressed, millis()) == Debouncer::Event::Pressed) {
    paused = !paused;
    ext.toggle();
  }
}

void sensorTask() { samples.pushOverwrite(analogRead(kPotPin)); }

void reportTask() {
  Serial.print("avg:");     Serial.print(average(samples));
  Serial.print(" paused:"); Serial.println(paused ? 1 : 0);
}

Task tasks[] = {
  {500, 0, blinkTask},
  {5,   0, buttonTask},
  {20,  0, sensorTask},
  {200, 0, reportTask},
};

// พิมพ์ตัวเลข 50–5000 แล้ว Enter เพื่อเปลี่ยนรอบกระพริบ
void handleSerial() {
  static char buf[16];
  static size_t len = 0;
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (len > 0) {
        buf[len] = '\0';
        long ms = atol(buf);
        if (ms >= 50 && ms <= 5000) tasks[0].periodMs = ms;
        len = 0;
      }
    } else if (len < sizeof(buf) - 1) {
      buf[len++] = c;
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(kButtonPin, INPUT_PULLUP);
  status.begin();
  ext.begin();
  Serial.print("fw "); Serial.println(FW_VERSION);
}

void loop() {
  uint32_t now = millis();
  for (Task& t : tasks) {
    if (now - t.lastRunMs >= t.periodMs) {
      t.lastRunMs = now;
      t.run();
    }
  }
  handleSerial();
}

บรรทัด #if __has_include("src/version.h") ตรวจว่ามีไฟล์ version.h หรือไม่ ตอนนี้ยังไม่มี โปรแกรมจึงพิมพ์ fw dev ตอนเริ่ม ในบท 23 ระบบ CI จะสร้างไฟล์นี้พร้อมเลขเวอร์ชันจริงให้อัตโนมัติ

เช็กความเข้าใจ

ถ้าพิมพ์ข้อความยาวเกิน 15 ตัวอักษรจะเกิดอะไรขึ้น?

ตัวที่เกินจะถูกทิ้ง เพราะ len ไม่เพิ่มเกิน sizeof(buf) - 1 โปรแกรมไม่พังและไม่เขียนทับหน่วยความจำส่วนอื่น จากนั้นตัวเลขที่ได้จะถูกตรวจช่วงค่าอีกชั้น

ทำไมต้องเช็กทั้ง \n และ \r?

แต่ละโปรแกรมส่งการขึ้นบรรทัดไม่เหมือนกัน บางตัวส่ง \n บางตัวส่ง \r หรือส่งทั้งคู่ การรับทั้งสองแบบ และข้ามบรรทัดว่าง (len == 0) ทำให้ใช้ได้กับทุกโปรแกรม
