Reference, const และ constexpr แทน #define

25 นาที · ฮาร์ดแวร์: บอร์ดอย่างเดียว

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย เลิกใช้ #define กับค่าคงที่ และส่งพารามิเตอร์ด้วย reference ได้ถูกที่

C

ค่าคงที่ใช้ #define ไม่มี type ไม่มี scope แก้ตัวแปรของผู้เรียกต้องส่ง pointer

C++

constexpr มี type และคำนวณตอน compile ไม่กิน RAM ส่วน T& คือชื่อเล่นของตัวแปรเดิม ไม่มี null ไม่ต้องเขียน *

Hands-on

rgb_cycle.ino
#include <Adafruit_TinyUSB.h>   // เปิด USB และ Serial (core นี้ต้องมี)
constexpr uint8_t kLedPins[] = {LED_RED, LED_GREEN, LED_BLUE};
constexpr size_t  kLedCount  = sizeof(kLedPins) / sizeof(kLedPins[0]);
constexpr uint32_t kStepMs   = 150;

void allOff() {
  for (uint8_t pin : kLedPins) {   // range-based for (C++11)
    digitalWrite(pin, HIGH);
  }
}

void nextIndex(size_t& idx) {      // reference: แก้ตัวแปรของผู้เรียกได้เลย
  idx = (idx + 1) % kLedCount;
}

size_t current = 0;

void setup() {
  for (uint8_t pin : kLedPins) pinMode(pin, OUTPUT);
  allOff();
}

void loop() {
  allOff();
  digitalWrite(kLedPins[current], LOW);
  delay(kStepMs);
  nextIndex(current);
}

ลองเพิ่มฟังก์ชัน void printPin(const uint8_t& p) แล้วพยายามเขียน p = 0; ข้างใน ดู error ที่ compiler ฟ้อง นี่คือประโยชน์ของ const: จับบั๊กตอน compile

เช็กความเข้าใจ

เมื่อไหร่ควรส่ง const T& แทนส่งค่า?

เมื่อ T ใหญ่กว่า register (struct, buffer) เพื่อเลี่ยงการ copy ส่วน int, float, pointer ส่งค่าตรง ๆ เร็วกว่าและอ่านง่ายกว่า

constexpr ต่างจาก const อย่างไร?

const = ห้ามแก้ แต่อาจรู้ค่าตอน runtime ส่วน constexpr = ต้องรู้ค่าตอน compile จึงใช้เป็นขนาด array หรือ template argument ได้

มุมงานจริง

ใน code review ของทีม firmware ค่าคงที่แบบ #define มักถูกขอให้เปลี่ยนเป็น constexpr เพราะ #define ไม่มี type checking, ไม่เคารพ namespace/scope, debugger มองไม่เห็นชื่อ และอาจเกิด side effect จาก macro expansion
