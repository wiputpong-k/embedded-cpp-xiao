Interrupt, volatile และ lambda

25 นาที · ฮาร์ดแวร์: ปุ่มกดที่ D1

ในบทนี้
  1. Hands-on
  2. เช็กความเข้าใจ

เป้าหมาย เขียน ISR ที่ถูกต้อง และแชร์ข้อมูลระหว่าง ISR กับ loop อย่างปลอดภัย

Hands-on

isr_count.ino
#include <Adafruit_TinyUSB.h>   // เปิด USB และ Serial (core นี้ต้องมี)
constexpr uint8_t kButtonPin = D1;
volatile uint32_t pressCount = 0;   // volatile: ค่าเปลี่ยนนอก flow ปกติ ห้าม cache

void setup() {
  Serial.begin(115200);
  pinMode(kButtonPin, INPUT_PULLUP);
  // lambda ที่ไม่ capture อะไร แปลงเป็น function pointer ได้
  attachInterrupt(digitalPinToInterrupt(kButtonPin), [] {
    pressCount = pressCount + 1;
  }, FALLING);
}

void loop() {
  uint32_t count;
  noInterrupts();          // copy ออกมาแบบ atomic
  count = pressCount;
  interrupts();
  Serial.println(count);
  delay(500);
}

กดปุ่ม 1 ครั้ง ตัวเลขมักเพิ่มมากกว่า 1 นั่นคือ bounce ที่เห็นในบท 07 ข้อสรุป: ISR บนปุ่มกลไกต้อง debounce ด้วยเวลา (เช่นเช็ก millis() ใน ISR หรือส่ง event ให้ loop จัดการ)

เช็กความเข้าใจ

volatile ทำให้ pressCount++ เป็น atomic ไหม?

ไม่ volatile แค่บังคับให้อ่าน/เขียน memory จริงทุกครั้ง แต่ ++ คือ load-add-store 3 คำสั่ง ถูกแทรกกลางได้ ต้องใช้ critical section หรือ std::atomic

อะไรห้ามทำใน ISR?

delay(), Serial.print(), malloc/new, ทำงานนาน และเรียกฟังก์ชันที่ไม่ reentrant ให้ ISR ตั้ง flag หรือ push ลง buffer แล้วกลับเร็วที่สุด

มุมงานจริง

การลืม volatile เป็นสาเหตุอันดับต้น ๆ ของอาการ "build debug ทำงาน แต่ build release พัง" จำ 3 กรณีที่ต้องใช้: ตัวแปรที่ ISR แก้, memory-mapped register, ตัวแปรที่แชร์ระหว่าง thread และจำไว้ว่ามันไม่ใช่ synchronization
